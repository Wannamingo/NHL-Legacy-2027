cmake_minimum_required(VERSION 3.25)
project(nhl_host_codegen LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(NHL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(REXGLUE_ROOT "${NHL_ROOT}/third_party/rexglue-sdk")
set(REX_HOST_SDK "${NHL_ROOT}/private/host-sdk/release-0.8.0/win-amd64"
    CACHE PATH "Authenticated official ReXGlue host runtime")
find_package(Threads REQUIRED)
find_package(rexglue CONFIG REQUIRED PATHS "${REX_HOST_SDK}" NO_DEFAULT_PATH)
add_library(rexruntime INTERFACE)
target_link_libraries(rexruntime INTERFACE rex::runtime)
add_library(rexcore INTERFACE)
target_link_libraries(rexcore INTERFACE rex::runtime)
add_library(tomlplusplus ALIAS rex::tomlplusplus)
add_library(simde ALIAS rex::simde)
add_library(disasm ALIAS rex::disasm)
add_library(xxHash::xxhash ALIAS rex::xxhash)
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN REXGLUE_BUILD_CONFIG="Release")
add_compile_options(-fno-char8_t -fno-strict-aliasing -msse4.1)
include_directories("${REXGLUE_ROOT}/include")
set(REXGLUE_ENABLE_TRACY OFF)
set(REXGLUE_ENABLE_PERF_COUNTERS OFF)
add_subdirectory("${REXGLUE_ROOT}/src/codegen" codegen)
add_subdirectory("${REXGLUE_ROOT}/src/rexglue" cli)
target_compile_definitions(rexglue PRIVATE REXGLUE_VERSION_FLOOR="0.8.1")
target_sources(rexglue PRIVATE
  "${REXGLUE_ROOT}/src/core/platform/console_win.cpp"
  "${REXGLUE_ROOT}/src/core/platform/env_win.cpp")
file(GLOB runtime_dlls "${REX_HOST_SDK}/bin/*.dll")
file(COPY ${runtime_dlls} DESTINATION "${CMAKE_CURRENT_BINARY_DIR}/cli")
